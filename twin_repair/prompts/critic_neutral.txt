import json

with open("/workspace/public_state.json", "r", encoding="utf-8") as f:
    pub = json.load(f)

sld_img_name = str(pub.get("current_sld_image", "current_sld.png"))
target_revision = "UNKNOWN"  # this drawing carries no revision letter; never guess one

# 1. What you visibly observe in Image 2 (render0_wiring.png) — blue cable/badge = "F1", orange cable/badge = "F2":
observed_in_render0 = {
    "N1": {"feeder_id": "<F1_OR_F2>", "bbox_xyxy_px": [180, 150, 360, 420]},
    "N2": {"feeder_id": "<F1_OR_F2>", "bbox_xyxy_px": [410, 150, 610, 420]},
    "N3": {"feeder_id": "<F1_OR_F2>", "bbox_xyxy_px": [660, 150, 870, 420]},
    "N4": {"feeder_id": "<F1_OR_F2>", "bbox_xyxy_px": [910, 150, 1120, 420]},
}

# 2. What you visibly observe in Image 1 (current SLD) — filled black dot (●) on F1 (y=210) vs F2 (y=290):
target_in_sld = {
    "N1": {"feeder_id": "<F1_OR_F2>", "bbox_xyxy_px": [150, 200, 270, 420]},

    "N3": {"feeder_id": "<F1_OR_F2>", "bbox_xyxy_px": [670, 200, 790, 420]},
    "N4": {"feeder_id": "<F1_OR_F2>", "bbox_xyxy_px": [930, 200, 1050, 420]},
}

racks_ordered = ("N1", "N2", "N3", "N4")
observed_render_edges = [
    {
        "rack_id": rk,
        "feeder_id": observed_in_render0[rk]["feeder_id"],
        "image_ref": "render0_wiring.png",
        "bbox_xyxy_px": observed_in_render0[rk]["bbox_xyxy_px"],
    }
    for rk in racks_ordered
]

    {

        "feeder_id": target_in_sld[rk]["feeder_id"],
        "image_ref": sld_img_name,
        "bbox_xyxy_px": target_in_sld[rk]["bbox_xyxy_px"],
    }

                "op": "replace_feeder",
                "rack_id": rk,
                "from": obs_f,
                "to": req_f,
            }
        )

decision = "repair" if len(operations) > 0 else "no_change"
patch = {
    "schema_version": "repair_patch.v1",
    "case_id": str(pub["case_id"]),
    "basis_scene_sha256": str(pub["basis_scene_sha256"]),

    "observed_render_edges": observed_render_edges,
    "target_sld_edges": target_sld_edges,
    "discrepancies": discrepancies,
    "operations": operations,
    "confidence": 0.94,
    "decision": decision,
}

with open("/workspace/repair_patch.json", "w", encoding="utf-8") as f:
    json.dump(patch, f, indent=2, sort_keys=True)
print("Wrote /workspace/repair_patch.json with decision:", decision)
